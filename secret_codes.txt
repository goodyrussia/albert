#!/usr/bin/env bash
MASTER_KEY="TECHFEEDS-VPN-V1.2-EXCLUSIVE-MASTER-KEY-2026"

generate_v12_code() {
  local DURATION=$(echo "$1" | tr '[:lower:]' '[:upper:]')
  local SERIAL=$(head -c 32 /dev/urandom | LC_ALL=C tr -dc 'A-Z0-9' | head -c 4)
  local PAYLOAD="TFV12|${DURATION}|${SERIAL}"
  local SIG=$(printf "%s" "$PAYLOAD" | openssl dgst -sha256 -hmac "$MASTER_KEY" | awk '{print $NF}' | cut -c1-8 | tr '[:lower:]' '[:upper:]')
  echo "TFV12-${DURATION}-${SERIAL}-${SIG}"
}

echo "15 Minutes : $(generate_v12_code 15M)"

echo "7 Days     : $(generate_v12_code 7D)"
echo "1 Month    : $(generate_v12_code 1MO)"
echo "12 Months  : $(generate_v12_code 12MO)"
